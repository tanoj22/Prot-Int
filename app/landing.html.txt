const canvas=document.getElementById('bg');
const renderer=new THREE.WebGLRenderer({canvas,antialias:true,alpha:true});
renderer.setPixelRatio(Math.min(window.devicePixelRatio,2));
renderer.setSize(window.innerWidth,window.innerHeight);
const scene=new THREE.Scene();
const camera=new THREE.PerspectiveCamera(55,window.innerWidth/window.innerHeight,0.1,1000);
camera.position.set(0,0,30);
const ambLight=new THREE.AmbientLight(0x334455,0.6);
scene.add(ambLight);

scene.add(dirLight);
const dirLight2=new THREE.DirectionalLight(0x10B981,0.5);
dirLight2.position.set(-5,-5,5);
scene.add(dirLight2);
const proteinGroup=new THREE.Group();
scene.add(proteinGroup);
const nRes=80;
const helixR=3.5;
const helixP=0.35;
const helixY=0.18;
const points=[];
for(let i=0;i<nRes;i++){
const t=i*helixP;
const x=helixR*Math.cos(t);
const z=helixR*Math.sin(t);
const y=(i-nRes/2)*helixY;
points.push(new THREE.Vector3(x,y,z));
}
for(let i=0;i<points.length;i++){
const frac=i/points.length;
const col=new THREE.Color().lerpColors(new THREE.Color(0x22D3EE),new THREE.Color(0x34D399),frac);
const geo=new THREE.SphereGeometry(0.28,12,12);
const mat=new THREE.MeshPhongMaterial({color:col,emissive:col,emissiveIntensity:0.2,shininess:80});
const sphere=new THREE.Mesh(geo,mat);
sphere.position.copy(points[i]);
proteinGroup.add(sphere);
if(i>0){
const dir=new THREE.Vector3().subVectors(points[i],points[i-1]);
const len=dir.length();
const cGeo=new THREE.CylinderGeometry(0.08,0.08,len,6);
const cMat=new THREE.MeshPhongMaterial({color:col,emissive:col,emissiveIntensity:0.1,transparent:true,opacity:0.6});
const cyl=new THREE.Mesh(cGeo,cMat);
const mid=new THREE.Vector3().addVectors(points[i],points[i-1]).multiplyScalar(0.5);
cyl.position.copy(mid);
cyl.lookAt(points[i]);
cyl.rotateX(Math.PI/2);
proteinGroup.add(cyl);
}
}
const sideChainEvery=4;
for(let i=0;i<points.length;i+=sideChainEvery){
const frac=i/points.length;
const col=new THREE.Color().lerpColors(new THREE.Color(0x22D3EE),new THREE.Color(0x34D399),frac);
const center=new THREE.Vector3(0,points[i].y,0);
const outDir=new THREE.Vector3().subVectors(points[i],center).normalize();
const scLen=0.8+Math.random()*1.2;
const endPt=new THREE.Vector3().copy(points[i]).add(outDir.clone().multiplyScalar(scLen));
const sGeo=new THREE.SphereGeometry(0.18,8,8);
const sMat=new THREE.MeshPhongMaterial({color:col,emissive:col,emissiveIntensity:0.2,transparent:true,opacity:0.7});
const sSphere=new THREE.Mesh(sGeo,sMat);
sSphere.position.copy(endPt);
proteinGroup.add(sSphere);
const bDir=new THREE.Vector3().subVectors(endPt,points[i]);
const bLen=bDir.length();
const bGeo=new THREE.CylinderGeometry(0.04,0.04,bLen,4);
const bMat=new THREE.MeshPhongMaterial({color:col,transparent:true,opacity:0.3});
const bCyl=new THREE.Mesh(bGeo,bMat);
const bMid=new THREE.Vector3().addVectors(points[i],endPt).multiplyScalar(0.5);
bCyl.position.copy(bMid);
bCyl.lookAt(endPt);
bCyl.rotateX(Math.PI/2);
proteinGroup.add(bCyl);
}
const pCount=300;
const pGeo=new THREE.BufferGeometry();
const pPos=new Float32Array(pCount*3);
const pVel=new Float32Array(pCount*3);
for(let i=0;i<pCount;i++){
pPos[i*3]=(Math.random()-0.5)*60;
pPos[i*3+1]=(Math.random()-0.5)*40;
pPos[i*3+2]=(Math.random()-0.5)*40;
pVel[i*3]=(Math.random()-0.5)*0.005;
pVel[i*3+1]=(Math.random()-0.5)*0.005;
pVel[i*3+2]=(Math.random()-0.5)*0.005;
}
pGeo.setAttribute('position',new THREE.BufferAttribute(pPos,3));
const pMat=new THREE.PointsMaterial({color:0x22D3EE,size:0.08,transparent:true,opacity:0.3});
const particleSys=new THREE.Points(pGeo,pMat);
scene.add(particleSys);
let mouseX=0,mouseY=0;
document.addEventListener('mousemove',(e)=>{
mouseX=(e.clientX/window.innerWidth-0.5)*2;
mouseY=(e.clientY/window.innerHeight-0.5)*2;
});
let scrollY=0;
window.addEventListener('scroll',()=>{scrollY=window.scrollY});
window.addEventListener('resize',()=>{
camera.aspect=window.innerWidth/window.innerHeight;
camera.updateProjectionMatrix();
renderer.setSize(window.innerWidth,window.innerHeight);
});
function animate(){
requestAnimationFrame(animate);
const t=performance.now()*0.001;
proteinGroup.rotation.y=t*0.15+mouseX*0.3;
proteinGroup.rotation.x=Math.sin(t*0.1)*0.1+mouseY*0.2;
proteinGroup.position.y=0;
const pos=pGeo.attributes.position.array;
for(let i=0;i<pCount;i++){
pos[i*3]+=pVel[i*3];
pos[i*3+1]+=pVel[i*3+1];
pos[i*3+2]+=pVel[i*3+2];
if(Math.abs(pos[i*3])>30)pVel[i*3]*=-1;
if(Math.abs(pos[i*3+1])>20)pVel[i*3+1]*=-1;
if(Math.abs(pos[i*3+2])>20)pVel[i*3+2]*=-1;
}
pGeo.attributes.position.needsUpdate=true;
const fade=Math.max(0,1-scrollY/600);
renderer.domElement.style.opacity=0.4+fade*0.6;
renderer.render(scene,camera);
}
animate();
